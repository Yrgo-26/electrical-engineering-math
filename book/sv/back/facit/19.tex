\begin{facit}

\svar{1} $I_1 = 3\,\text{mA}$, $I_2 = 2\,\text{mA}$

\svar{2} \sv{a} $R = 2\,\Omega$ eller $R = 18\,\Omega$
\sv{b} $I = 3\,\text{A}$ respektive $1\,\text{A}$. $R = 18\,\Omega$ är att föredra: samma effekt
med en tredjedel av strömmen, och förlusten i den inre resistansen blir $6\,\text{W}$ i stället
för $54\,\text{W}$.

\svar{3} \sv{a} $U(I) = \frac{4(I + 3)}{I + 2}$, där $I \neq -2$ och $I \neq 3$
\sv{b} $U(2) = 5\,\text{V}$

\svar{4} \sv{a} $u(3) \approx 6{,}43\,\text{V}$
\sv{b} $0 \leq t \leq 10\,\text{s}$ och $2{,}30\,\text{V} \leq u(t) \leq 10\,\text{V}$
\sv{c} Efter cirka $6{,}23\,\text{s}$.

\svar{5} \sv{a} $u(t) = 5\sin\bigl(200\pi t + \frac{\pi}{2}\bigr)\,\text{V}$
\sv{b} Periodtiden är $10\,\text{ms}$. Kurvan börjar i $5\,\text{V}$, är noll vid $2{,}5$ och
$7{,}5\,\text{ms}$ och når $-5\,\text{V}$ vid $5\,\text{ms}$.

\svar{6} \sv{a} $P'(I) = 24 - 12I$ \sv{b} $I = 2\,\text{A}$
\sv{c} Maximum, eftersom $P''(2) = -12 < 0$.
\sv{d} Största värdet är $P(2) = 24\,\text{W}$.
\sv{e} En nedåtöppen parabel med maximipunkten $(2; 24)$. Den skär $I$-axeln i $(0; 0)$ och
$(4; 0)$.
\sv{f} $U = 12\,\text{V}$ och $R = 6\,\Omega$, alltså $R = R_{\text{i}}$ (effektanpassning).

\svar{7} \sv{a} $I(t) = 0{,}01t^3 + 0{,}05t^2 + 0{,}5t + C$
\sv{b} $q(t) = 0{,}01t^3 + 0{,}05t^2 + 0{,}5t$
\sv{c} $q_{\text{tot}}(t) = 0{,}01t^3 + 0{,}05t^2 + 0{,}5t + 1$
\sv{d} Exakt $4{,}44\,\text{C}$

\svar{8} \sv{a} $I = -0{,}25 - j1{,}75\,\text{A} \approx 1{,}8\angle{-1{,}7}\,\text{A}$, med vinkeln
i radianer ($-98{,}1^{\circ}$).
\sv{b} $R = 2\,\Omega$ och $X = 2\,\Omega$, alltså induktiv.
\sv{c} $L = \frac{1}{50\pi}\,\text{H} \approx 6{,}37\,\text{mH}$

\svar{9} \sv{a} En pil från origo till $10 - j5$, i fjärde kvadranten.
\sv{b} $I \approx 11{,}2e^{-j0{,}46}\,\text{mA}$: $\abs{I} = \sqrt{125} \approx 11{,}2\,\text{mA}$
och $\delta \approx -0{,}46\,\text{rad}$ ($-26{,}6^{\circ}$)
\sv{c} $w = 20\pi \approx 62{,}8\,\text{rad/s}$
\sv{d} $i(t) \approx 11{,}2e^{j(20\pi t - 0{,}46)}\,\text{mA}$

\svar{10} \sv{a} $U_1 \approx 1{,}81 + j0{,}85\,\text{V}$, $U_2 \approx 3{,}54 - j3{,}54\,\text{V}$,
$U_3 \approx 1{,}21 + j0{,}88\,\text{V}$
\sv{b} $U_{\text{tot}} \approx 6{,}56 - j1{,}81\,\text{V}$
\sv{c} $U_1$ och $U_3$ i första kvadranten, $U_2$ och $U_{\text{tot}}$ i fjärde; $U_{\text{tot}}$
ligger strax under den reella axeln.
\sv{d} $u_{\text{tot}}(t) \approx 6{,}81\sin(wt - 15{,}4^{\circ})\,\text{V}$

\end{facit}
