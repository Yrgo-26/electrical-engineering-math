\facitavsnitt{ot}{Övningstentamen 1}{Hela kursen}
\begin{facit}

\svar{1} $I_1 = 3\,\text{mA}$, $I_2 = 2\,\text{mA}$

\svar{2} \sv{a} $R = 2\,\Omega$ eller $R = 18\,\Omega$
\sv{b} $I = 3\,\text{A}$ respektive $1\,\text{A}$. $R = 18\,\Omega$ är att föredra: samma effekt
med en tredjedel av strömmen, och förlusten i den inre resistansen blir $6\,\text{W}$ i stället
för $54\,\text{W}$.

\svar{3} \sv{a} $U(I) = \frac{4(I + 3)}{I + 2}$, där $I \neq -2$ och $I \neq 3$
\sv{b} $U(2) = 5\,\text{V}$

\svar{4} \sv{a} $u(3) \approx 6{,}43\,\text{V}$
\sv{b} $0 \leq t \leq 10\,\text{s}$ och $2{,}30\,\text{V} \leq u(t) \leq 10\,\text{V}$
\sv{c} Efter cirka $6{,}23\,\text{s}$.

\svar{5} \sv{a} $u(t) = 5\sin\bigl(200\pi t + \frac{\pi}{2}\bigr)\,\text{V}$
\sv{b} Periodtiden är $10\,\text{ms}$. Kurvan börjar i $5\,\text{V}$, är noll vid $2{,}5$ och
$7{,}5\,\text{ms}$ och når $-5\,\text{V}$ vid $5\,\text{ms}$.

\svar{6} \sv{a} $P'(I) = 24 - 12I$ \sv{b} $I = 2\,\text{A}$
\sv{c} Maximum, eftersom $P''(2) = -12 < 0$.
\sv{d} Största värdet är $P(2) = 24\,\text{W}$.
\sv{e} En nedåtöppen parabel med maximipunkten $(2; 24)$. Den skär $I$-axeln i $(0; 0)$ och
$(4; 0)$.
\sv{f} $U = 12\,\text{V}$ och $R = 6\,\Omega$, alltså $R = R_{\text{i}}$ (effektanpassning).

\svar{7} \sv{a} $I(t) = 0{,}01t^3 + 0{,}05t^2 + 0{,}5t + C$
\sv{b} $q(t) = 0{,}01t^3 + 0{,}05t^2 + 0{,}5t$
\sv{c} $q_{\text{tot}}(t) = 0{,}01t^3 + 0{,}05t^2 + 0{,}5t + 1$
\sv{d} Exakt $4{,}44\,\text{C}$

\svar{8} \sv{a} $I = -0{,}25 - j1{,}75\,\text{A} \approx 1{,}8\angle{-1{,}7}\,\text{A}$, med vinkeln
i radianer ($-98{,}1^{\circ}$).
\sv{b} $R = 2\,\Omega$ och $X = 2\,\Omega$, alltså induktiv.
\sv{c} $L = \frac{1}{50\pi}\,\text{H} \approx 6{,}37\,\text{mH}$

\svar{9} \sv{a} En pil från origo till $10 - j5$, i fjärde kvadranten.
\sv{b} $I \approx 11{,}2e^{-j0{,}46}\,\text{mA}$: $\abs{I} = \sqrt{125} \approx 11{,}2\,\text{mA}$
och $\delta \approx -0{,}46\,\text{rad}$ ($-26{,}6^{\circ}$)
\sv{c} $w = 20\pi \approx 62{,}8\,\text{rad/s}$
\sv{d} $i(t) \approx 11{,}2e^{j(20\pi t - 0{,}46)}\,\text{mA}$

\svar{10} \sv{a} $U_1 \approx 1{,}81 + j0{,}85\,\text{V}$, $U_2 \approx 3{,}54 - j3{,}54\,\text{V}$,
$U_3 \approx 1{,}21 + j0{,}88\,\text{V}$
\sv{b} $U_{\text{tot}} \approx 6{,}56 - j1{,}81\,\text{V}$
\sv{c} $U_1$ och $U_3$ i första kvadranten, $U_2$ och $U_{\text{tot}}$ i fjärde; $U_{\text{tot}}$
ligger strax under den reella axeln.
\sv{d} $u_{\text{tot}}(t) \approx 6{,}81\sin(wt - 15{,}4^{\circ})\,\text{V}$

\end{facit}

\facitavsnitt{ot2}{Övningstentamen 2}{Hela kursen}
\begin{facit}

\svar{1} $I_1 = 2\,\text{mA}$, $I_2 = 4\,\text{mA}$

\svar{2} \sv{a} $U = 4\,\text{V}$ eller $U = 16\,\text{V}$
\sv{b} $U = 16\,\text{V}$: $I = 1\,\text{A}$, $R = 16\,\Omega$. $U = 4\,\text{V}$: $I = 4\,\text{A}$,
$R = 1\,\Omega$. $U = 16\,\text{V}$ är att föredra: samma effekt med en fjärdedel av strömmen.

\svar{3} \sv{a} $U(I) = \frac{3(I + 2)}{I + 3}$, där $I \neq \pm 3$ \sv{b} $I = 7\,\text{A}$

\svar{4} \sv{a} $RC = 10\,\text{s}$ \sv{b} $U_0 = 6{,}0 \cdot e^{0{,}5} \approx 9{,}89\,\text{V}$
\sv{c} $t = 10 \ln 2 \approx 6{,}93\,\text{s}$, oberoende av $U_0$.

\svar{5} \sv{a} Amplituden $325\,\text{V}$, effektivvärdet $\approx 230\,\text{V}$, $f = 50\,\text{Hz}$,
$T = 20\,\text{ms}$ och fasen $30^{\circ}$.
\sv{b} Kurvan startar i $162{,}5\,\text{V}$, har maximum vid $3{,}33\,\text{ms}$, noll vid
$8{,}33$ och $18{,}33\,\text{ms}$ och minimum vid $13{,}33\,\text{ms}$.

\svar{6} \sv{a} $i'(t) = 3t^2 - 18t + 24$ \sv{b} $t = 2\,\text{s}$ och $t = 4\,\text{s}$
\sv{c} Maximum vid $t = 2$ ($i''(2) = -6$), minimum vid $t = 4$ ($i''(4) = 6$).
\sv{d} $i(2) = 20\,\text{mA}$, $i(4) = 16\,\text{mA}$
\sv{e} Grafen startar i origo, stiger till $(2; 20)$, sjunker till $(4; 16)$ och stiger till
$(5; 20)$.
\sv{f} $u(1) = 0{,}5 \cdot 0{,}009 = 4{,}5\,\text{mV}$

\svar{7} \sv{a} $I(t) = 0{,}02t^3 + 0{,}2t^2 + t + C$
\sv{b} $q(t) = 0{,}02t^3 + 0{,}2t^2 + t$ \sv{c} $9{,}54\,\text{C}$ \sv{d} $3{,}18\,\text{A}$

\svar{8} \sv{a} $Z = 30 + j40 \approx 50\angle 53{,}1^{\circ}\,\Omega$
\sv{b} $I \approx 2\angle{-53{,}1^{\circ}}\,\text{A} = 1{,}2 - j1{,}6\,\text{A}$
\sv{c} $L = \frac{2}{5\pi}\,\text{H} \approx 127\,\text{mH}$

\svar{9} \sv{a} En pil från origo till $(-8; -6)$, i tredje kvadranten.
\sv{b} $I \approx 10e^{-j2{,}50}\,\text{mA}$ ($\delta \approx -143{,}1^{\circ}$)
\sv{c} $\omega = 100\pi\,\text{rad/s}$ \sv{d} $i(t) \approx 10\sin(100\pi t - 2{,}50)\,\text{mA}$

\svar{10} \sv{a} $U_1 \approx 2{,}00 + j3{,}46\,\text{V}$, $U_2 \approx 2{,}60 - j1{,}50\,\text{V}$,
$U_3 \approx -1{,}73 + j1{,}00\,\text{V}$
\sv{b} $U_{\text{tot}} \approx 2{,}87 + j2{,}96\,\text{V}$
\sv{c} $U_1$ i första, $U_2$ i fjärde och $U_3$ i andra kvadranten; $U_{\text{tot}}$ i första.
\sv{d} $u_{\text{tot}}(t) \approx 4{,}12\sin(wt + 46{,}0^{\circ})\,\text{V}$

\end{facit}
