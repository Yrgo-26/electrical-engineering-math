\section{Sammanfattning}
\label{c14:sec:sammanfattning}

\begin{booktable}{@{}p{12.5em}L@{}}
  \thead{Begrepp} & \thead{Formel} \\
  \midrule
  Primitiv funktion & $F'(x) = f(x)$ \\
  Obestämd integral & $\int f(x)\,dx = F(x) + C$ \\
  Bestämd integral & $\int_a^b f(x)\,dx = F(b) - F(a)$ \\
  $\int x^n\,dx$ & $\frac{x^{n+1}}{n+1} + C$ \\
  $\int e^{kx}\,dx$ & $\frac{1}{k}e^{kx} + C$ \\
  Laddning & $q(t) = \int i(t)\,dt$ \\
\end{booktable}

\needspace{6\baselineskip}
\subsection*{Det här ska du kunna}
Efter det här kapitlet ska du:
\begin{itemize}
  \item Förstå vad en integral är.
  \item Förstå begreppet primitiv funktion och kunna ta fram den för enkla funktioner.
  \item Kunna utföra enkla integreringar.
\end{itemize}

\testadigsjalv
\begin{enumerate}
  \item Beräkna $\displaystyle\int_0^2 (3x^2 + 1)\,dx$.
  \item Bestäm den primitiva funktionen $F(x)$ till $f(x) = 4x + e^x$. Ange
    integrationskonstanten $C$ i svaret.
\end{enumerate}
