\section{Primitiv funktion}
\label{c14:sec:primitiv}
En \textbf{primitiv funktion} $F(x)$ till $f(x)$ uppfyller
\[
  F'(x) = f(x).
\]
Det finns oändligt många primitiva funktioner, och de skiljer sig åt med en konstant $C$:
\[
  \int f(x)\,dx = F(x) + C
\]
$C$ kallas \textbf{integrationskonstanten}.

\section{Grundläggande integreringsregler}
\label{c14:sec:regler}
Tabellen visar de primitiva funktionerna till de vanligaste funktionerna.

{\renewcommand{\arraystretch}{1.9}%
\begin{narrowtable}{@{}l@{\hspace{3em}}l@{}}
  \thead{Funktion $f(x)$} & \thead{Primitiv funktion $F(x)$} \\
  \midrule
  $x^n$ \enspace ($n \neq -1$) & $\dfrac{x^{n+1}}{n+1} + C$ \\
  $e^x$ & $e^x + C$ \\
  $e^{kx}$ & $\dfrac{1}{k}e^{kx} + C$ \\
  $\dfrac{1}{x}$ & $\ln\abs{x} + C$ \\
  $\sin(x)$ & $-\cos(x) + C$ \\
  $\cos(x)$ & $\sin(x) + C$ \\
  $\sin(kx)$ & $-\dfrac{1}{k}\cos(kx) + C$ \\
  $\cos(kx)$ & $\dfrac{1}{k}\sin(kx) + C$ \\
\end{narrowtable}}

Dessutom gäller \textbf{linjäritetsegenskapen}:
\[
  \int \bigl[af(x) + bg(x)\bigr]\,dx = a\int f(x)\,dx + b\int g(x)\,dx
\]

\section{Bestämd integral och area}
\label{c14:sec:bestamd}
Den \textbf{bestämda integralen} ger arean under kurvan $y = f(x)$ från $a$ till $b$:
\[
  \int_a^b f(x)\,dx = F(b) - F(a)
\]

\begin{aside}
\note[Obs] Om $f(x) < 0$ i delar av intervallet bidrar de delarna med negativ area.
\end{aside}

\section{Integrationskonstanten och startvillkor}
\label{c14:sec:startvillkor}
För en primitiv funktion med ett känt startvillkor $F(x_0) = y_0$ bestäms $C$ av
\[
  C = y_0 - F_{\text{utan }C}(x_0),
\]
där $F_{\text{utan }C}$ är den primitiva funktionen skriven utan konstanten.

\section{Tillämpning: laddning från ström}
\label{c14:sec:laddning}
Laddningen $q(t)$ ges av integralen av strömmen $i(t)$:
\[
  q(t) = \int_0^t i(\tau)\,d\tau + q(0)
\]

\section{Typexempel}
\label{c14:sec:typexempel}

\begin{exempel}[Beräkna area]
Beräkna arean under $f(x) = 2x + 1$ för $0 \leq x \leq 2$.
\[
  \int_0^2 (2x + 1)\,dx = \Bigl[x^2 + x\Bigr]_0^2 = (4 + 2) - 0 = 6
\]
\end{exempel}

\begin{exempel}[Area under parabel]
Beräkna arean under $f(x) = x^2$ för $0 \leq x \leq 3$.
\[
  \int_0^3 x^2\,dx = \left[\frac{x^3}{3}\right]_0^3 = \frac{27}{3} - 0 = 9
\]
\end{exempel}

\begin{exempel}[Laddning via integrering]
Under tiden $0 \leq t \leq 2\,\text{s}$ passerar strömmen $i(t) = (2t + 4)\,\text{A}$. Beräkna
laddningen $q(t)$ med $q(0) = 0$.
\[
  q(t) = \int_0^t (2\tau + 4)\,d\tau = \Bigl[\tau^2 + 4\tau\Bigr]_0^t = t^2 + 4t
\]
Den primitiva funktionen $I(t) = t^2 + 4t$ uppfyller startvillkoret, eftersom $I(0) = 0$. Efter
två sekunder är laddningen
\[
  q(2) = 4 + 8 = 12\,\text{C}.
\]
\end{exempel}

\begin{exempel}[Bestämd integral med negativa värden]
Beräkna $\displaystyle\int_{-1}^{2} (x^2 - 1)\,dx$.
\begin{align*}
  \left[\frac{x^3}{3} - x\right]_{-1}^{2}
    &= \left(\frac{8}{3} - 2\right) - \left(-\frac{1}{3} + 1\right) \\
    &= \frac{8}{3} - 2 + \frac{1}{3} - 1 = 3 - 3 = 0
\end{align*}
De positiva och negativa areorna tar ut varandra.
\end{exempel}
