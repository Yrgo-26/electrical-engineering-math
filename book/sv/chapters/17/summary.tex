\section{Sammanfattning}

\begin{booktable}{@{}p{8.5em}L@{}}
  \thead{Begrepp} & \thead{Beskrivning} \\
  \midrule
  Fasor & $U = \abs{U}\,\angle\,\delta$ representerar $u(t) = \abs{U}\sin(\omega t + \delta)$ \\
  Fasoraddition & Konvertera till rektangulär form, addera, konvertera tillbaka \\
  Multiplikation & $\abs{z_1}\abs{z_2}\,\angle\,(\delta_1 + \delta_2)$ \\
  Division & $(\abs{z_1}/\abs{z_2})\,\angle\,(\delta_1 - \delta_2)$ \\
  Impedans & $Z = U/I$ \\
\end{booktable}

sinusformade signaler.

    polär form.
