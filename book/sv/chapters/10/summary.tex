\section{Sammanfattning}
\label{c10:sec:sammanfattning}

\begin{booktable}{@{}p{14em}L@{}}
  \thead{Begrepp} & \thead{Formel} \\
  \midrule
  Exponentiell ekvation & $a^x = b \;\Rightarrow\; x = \frac{\log b}{\log a}$ \\
  $\log(a^n)$ & $n \cdot \log a$ \\
  Spänningsförstärkning (dB) &
    $G_{\text{dB}} = 20\log\left(\frac{U_{\text{ut}}}{U_{\text{in}}}\right)$ \\
  dB till linjär förstärkning & $G_{\text{lin}} = 10^{G_{\text{dB}}/20}$ \\
\end{booktable}

\needspace{6\baselineskip}
\subsection*{Det här ska du kunna}
Efter det här kapitlet ska du:
\begin{itemize}
  \item Kunna räkna och lösa ekvationer med logaritmer och exponentialfunktioner.
  \item Känna till decibel.
\end{itemize}

\testadigsjalv
\begin{enumerate}
  \item Lös $4^x = 64$.
  \item En förstärkare har $G_{\text{dB}} = 40\,\text{dB}$. Beräkna den linjära
    spänningsförstärkningen $G_{\text{lin}}$.
\end{enumerate}
