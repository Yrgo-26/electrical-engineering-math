\section{Sammanfattning}
\label{c13:sec:sammanfattning}

\begin{narrowtable}{@{}ll@{}}
  \thead{Funktion} & \thead{Derivata} \\
  \midrule
  $e^{kx}$ & $ke^{kx}$ \\
  $\ln x$ & $1/x$ \\
  $\sin(kx)$ & $k\cos(kx)$ \\
  $\cos(kx)$ & $-k\sin(kx)$ \\
  $g(x) \cdot h(x)$ & $g'(x) \cdot h(x) + g(x) \cdot h'(x)$ (produktregeln) \\
  $g(u(x))$ & $g'(u(x)) \cdot u'(x)$ (kedjeregeln) \\
\end{narrowtable}
