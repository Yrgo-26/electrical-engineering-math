\section{Sammanfattning}
\label{c12:sec:sammanfattning}

\begin{booktable}{@{}p{12.5em}L@{}}
  \thead{Begrepp} & \thead{Formel eller regel} \\
  \midrule
  Derivata av $x^n$ & $nx^{n-1}$ \\
  Stationär punkt & $f'(x_0) = 0$ \\
  Maximum & $f'(x_0) = 0$ och $f''(x_0) < 0$ \\
  Minimum & $f'(x_0) = 0$ och $f''(x_0) > 0$ \\
  Tangentens ekvation & $y = f'(x_0)(x - x_0) + f(x_0)$ \\
\end{booktable}

\needspace{6\baselineskip}
\subsection*{Det här ska du kunna}
Efter det här kapitlet ska du:
\begin{itemize}
  \item Förstå hur derivatan hänger ihop med tangenten, räta linjens ekvation och maximi- och
    minimipunkter.
  \item Kunna derivera polynomfunktioner.
\end{itemize}

\testadigsjalv
\begin{enumerate}
  \item Derivera $f(x) = 4x^3 - 2x^2 + 7$.
  \item Bestäm extrempunktens koordinater för $f(x) = x^2 - 2x - 8$ och avgör om det är ett
    maximum eller ett minimum.
\end{enumerate}
