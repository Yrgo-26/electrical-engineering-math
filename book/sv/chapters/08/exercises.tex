\section{Uppgifter}
\label{c8:sec:uppgifter}

\begin{aside}
\textbf{Så arbetar du med uppgifterna.} Del~1 och del~2 räknas under lektionen: först på egen
hand, några minuter per uppgift, sedan gemensamt. Del~3 är extrauppgifter att träna vidare på
efter lektionen. De flesta går att räkna i huvudet eller med papper och penna.

\facitnotis{L08}
\end{aside}

% ------------------------------------------------------------------------------------------
\uppgiftsdel{Del 1}{Funktionsbegrepp, definitions- och värdemängd}

\uppgift{1.1}{Är sambandet en funktion?}
Avgör om strömmen $I$ är en funktion av spänningen $U$. Motivera ditt svar.
\begin{parts}
  \item $I = \dfrac{U}{10}$ (Ohms lag för ett $10\,\Omega$-motstånd)
  \item Mätparen $(U, I)$ med $U$ i V och $I$ i mA: $\lbrace(2, 100),\ (4, 200),\ (6, 250),\ (8,
    250)\rbrace$
  \item Mätparen $(U, I)$ med $U$ i V och $I$ i mA: $\lbrace(1, 2),\ (1, 5),\ (2, 3)\rbrace$
\end{parts}

\uppgift{1.2}{Definitions- och värdemängd}
Bestäm definitionsmängden och värdemängden.
\begin{parts}
  \item Temperatursensorn \code{TMP36} ger utspänningen $U(T) = 0{,}01T + 0{,}5$ (i V) vid
    temperaturen $T$ (i \textdegree C). Sensorn fungerar för $-40 \leq T \leq 125$.
  \item Spänningen över ett $100\,\Omega$-motstånd som utvecklar effekten $P$ (i W) är $U(P) =
    \sqrt{100P}$.
  \item Effekten i ett $10\,\Omega$-motstånd är $P(U) = \dfrac{U^2}{10}$, där spänningen kan vara $-20
    \leq U \leq 20$ (i V).
\end{parts}

\uppgift{1.3}{Linjär resistans hos en temperaturberoende resistor}
Resistansen $R$ (i $\Omega$) hos en viss temperaturberoende resistor beror linjärt på temperaturen
$T$ (i $^{\circ}$C) enligt $R(T) = kT + m$. Vid $T = 0\,^{\circ}\text{C}$ är $R = 100\,\Omega$, och
vid $T = 50\,^{\circ}\text{C}$ är $R = 175\,\Omega$.
\begin{parts}
  \item Bestäm $k$ och $m$.
  \item Beräkna resistansen vid $T = 25\,^{\circ}\text{C}$.
\end{parts}

\uppgiftsdel{Del 2}{Funktionstyper och tillämpningar}

\uppgift{2.1}{Ett batteri som åldras}
Ett nytt litiumjonbatteri har kapaciteten $4000\,\text{mAh}$ och förlorar $3\,\%$ av kapaciteten
per år. Efter $x$ år är kapaciteten
\[
  C(x) = 4000 \cdot 0{,}97^x\,\text{mAh}.
\]
\begin{parts}
  \item Beräkna $C(10)$ och tolka svaret.
  \item Ett batteri brukar räknas som uttjänt när kapaciteten har sjunkit till $80\,\%$. Efter hur
    många år händer det?
\end{parts}

\uppgift{2.2}{Effekt som potensfunktion av strömmen}
Effekten $P$ (i W) som utvecklas i ett motstånd $R = 50\,\Omega$ beror på strömmen $I$ (i A)
enligt
\[
  P(I) = R \cdot I^2.
\]
\begin{parts}
  \item Vilken typ av funktion är $P(I)$?
  \item Beräkna $P(0{,}5)$ och $P(2)$.
  \item En säkring i kretsen begränsar strömmen till $0 \leq I \leq 3\,\text{A}$. Bestäm
    definitions- och värdemängd för $P(I)$ inom detta intervall.
\end{parts}

\uppgift{2.3}{Tolkning av mätvärden}
En spänning $U(t)$ mäts vid flera tidpunkter under en urladdning:

\begin{narrowtable}{@{}l*{5}{>{\centering\arraybackslash}p{2.2em}}@{}}
  $t$ (s) & $0$ & $1$ & $2$ & $3$ & $4$ \\
  \midrule
  $U(t)$ (V) & $12$ & $9$ & $6$ & $3$ & $0$ \\
\end{narrowtable}

\begin{parts}
  \item Visa utifrån tabellen att sambandet mellan $t$ och $U(t)$ är linjärt.
  \item Bestäm en formel $U(t) = kt + m$ utifrån tabellen.
  \item Om mönstret i tabellen fortsatte, vid vilken tidpunkt $t$ skulle $U(t) = -3\,\text{V}$?
    Är detta ett rimligt värde för en urladdningsspänning? Motivera varför den linjära modellen
    inte kan gälla för alla $t \geq 0$.
\end{parts}

\uppgiftsdel{Del 3}{Extrauppgifter}
Uppgifterna nedan är extra träning och görs med fördel på egen hand efter lektionen.

\uppgift{3.1}{Beräkna funktionsvärden}
Ett batteri har klämspänningen $U(I) = 12 - 2I$ (i V), ett motstånd utvecklar effekten $P(I) =
5I^2$ (i W) och strömmen genom en last är $I(R) = \dfrac{24}{R}$ (i A). Beräkna:
\begin{delar}(3)
  \task $U(1)$
  \task $U(4)$
  \task $P(0)$
  \task $P(3)$
  \task $I(8)$
  \task $I(12)$
\end{delar}

\uppgift{3.2}{Lös ekvationen $f(x) = c$}
Samma funktioner som i uppgift~3.1.
\begin{parts}
  \item Vid vilken ström är klämspänningen $9\,\text{V}$?
  \item Bestäm nollstället till $U(I)$. Vad innebär det för batteriet?
  \item Vid vilken ström utvecklar motståndet $20\,\text{W}$?
  \item Kan $P(I)$ bli negativ? Motivera.
  \item Vilken resistans ger strömmen $0{,}5\,\text{A}$?
\end{parts}

\uppgift{3.3}{Definitionsmängd}
Bestäm den matematiska definitionsmängden. Ange sedan vilka värden som är fysikaliskt rimliga.
\begin{delar}(2)
  \task $I(R) = \dfrac{10}{R}$
  \task $I(R) = \dfrac{12}{R + 2}$
  \task $U(P) = \sqrt{50P}$
  \task $X(f) = \dfrac{1000}{f}$
\end{delar}

\uppgift{3.4}{Värdemängd}
Bestäm värdemängden.
\begin{parts}
  \item $P(I) = 10I^2$ (i W) för $0 \leq I \leq 2$ (i A)
  \item $U(I) = 12 - 2I$ (i V) för $0 \leq I \leq 6$ (i A)
  \item En likriktare ger utspänningen $u_{\text{ut}} = \abs{u_{\text{in}}}$ för $-10 \leq
    u_{\text{in}} \leq 10$ (i V).
  \item $T(U) = 100U - 50$ (i \textdegree C) för $0{,}1 \leq U \leq 1{,}75$ (i V)
\end{parts}

\uppgift{3.5}{Linjär funktion ur två punkter}
Bestäm den linjära funktionen.
\begin{parts}
  \item Ett batteri har klämspänningen $9\,\text{V}$ vid $0\,\text{A}$ och $8\,\text{V}$ vid
    $2\,\text{A}$. Bestäm $U(I)$.
  \item En temperatursensor ger $0{,}5\,\text{V}$ vid $0\,^{\circ}\text{C}$ och $1{,}5\,\text{V}$ vid
    $100\,^{\circ}\text{C}$. Bestäm $U(T)$.
  \item Ett Pt100-element har resistansen $100\,\Omega$ vid $0\,^{\circ}\text{C}$ och
    $138{,}5\,\Omega$ vid $100\,^{\circ}\text{C}$. Bestäm $R(T)$.
  \item En laddström är $2\,\text{A}$ vid $t = 0\,\text{s}$ och $2\,\text{A}$ vid $t = 5\,\text{s}$.
    Bestäm $I(t)$.
\end{parts}

\uppgift{3.6}{Tolkning av $k$ och $m$}
Ett belastat batteri har klämspänningen $U(I) = 12 - 0{,}5I$ (i V), där $I$ är strömmen i A.
\begin{parts}
  \item Vad betyder $k$ grafiskt?
  \item Vad betyder $m$ grafiskt?
  \item Är $U(I)$ växande eller avtagande? Motivera.
  \item Bestäm nollstället till $U(I)$.
  \item Vad representerar $k = -0{,}5$ respektive $m = 12$ fysikaliskt?
\end{parts}

\uppgift{3.7}{Potensfunktioner}
Effekten i ett $10\,\Omega$-motstånd är $P(I) = 10I^2$ (i W).
\begin{parts}
  \item Beräkna $P(3)$ och $P(-3)$. Varför blir effekten densamma när strömmen byter riktning?
  \item Hur många gånger större blir effekten om strömmen tredubblas?
  \item Hur ändras effekten om strömmen halveras?
  \item Effekten kan också skrivas $P(U) = \dfrac{U^2}{10}$. Beräkna $P(5)$ och $P(10)$. Hur ändras
    effekten när spänningen fördubblas?
\end{parts}

\uppgift{3.8}{Exponentialfunktioner}
En förstärkarkedja fördubblar signalen i varje steg, så att $u(n) = 5 \cdot 2^n$ (i mV) efter $n$
steg. En dämpare halverar i stället signalen i varje steg, så att $g(n) = 100 \cdot 0{,}5^n$ (i
mV).
\begin{parts}
  \item Beräkna $u(0)$, $u(1)$ och $u(3)$.
  \item Beräkna $g(0)$, $g(1)$ och $g(3)$.
  \item Vilken av funktionerna är växande respektive avtagande? Vad är det som avgör?
  \item Vad är $C$ i respektive funktion, och vilken betydelse har det?
\end{parts}

\uppgift{3.9}{Förändringsfaktor}
\begin{parts}
  \item Elpriset ökar med $5\,\%$ per år. Vilken är förändringsfaktorn $a$?
  \item Ett batteris kapacitet minskar med $20\,\%$ per år. Vilken är $a$?
  \item En signal dämpas till $70\,\%$ av sitt värde för varje steg den passerar. Startamplituden är
    $2\,\text{V}$. Skriv en funktion för amplituden efter $n$ steg.
  \item Beräkna amplituden efter $3$ steg.
  \item Efter hur många steg understiger amplituden $1\,\text{V}$?
\end{parts}

\uppgift{3.10}{Linjärt eller exponentiellt?}
Två storheter har mätts upp en gång per år: $f(x)$ är antalet laddstolpar i en kommun (i
hundratal) och $g(x)$ är antalet uppkopplade sensorer i ett nät (i tusental), där $x$ är antal år.

\begin{narrowtable}{@{}l*{5}{>{\centering\arraybackslash}p{2.2em}}@{}}
  $x$ & $0$ & $1$ & $2$ & $3$ & $4$ \\
  \midrule
  $f(x)$ & $3$ & $5$ & $7$ & $9$ & $11$ \\
  $g(x)$ & $3$ & $6$ & $12$ & $24$ & $48$ \\
\end{narrowtable}

\begin{parts}
  \item Vilken av funktionerna är linjär? Motivera utifrån tabellen.
  \item Vilken är exponentiell? Motivera.
  \item Bestäm en formel för $f(x)$.
  \item Bestäm en formel för $g(x)$.
  \item Beräkna $f(10)$ och $g(10)$.
\end{parts}

\uppgift{3.11}{Nollställen}
Bestäm nollställena och tolka dem.
\begin{parts}
  \item Klämspänningen $U(I) = 12 - 4I$ (i V) för ett batteri, där $I$ är strömmen i A.
  \item Effekten $P(I) = 12I - 2I^2$ (i W) som samma sorts batteri, med inre resistansen $2\,\Omega$,
    levererar till en last.
  \item Spänningen $U(t) = 6 - 0{,}5t$ (i V) under en urladdning, där $t$ är tiden i s.
\end{parts}

\uppgift{3.12}{Spänningsdelare som funktion}
En spänningsdelare består av $R_1 = 1\,\text{k}\Omega$ och en potentiometer med resistansen $x$
(i k$\Omega$). Utspänningen ges av
\[
  U(x) = 12 \cdot \frac{x}{1 + x}\,\text{V}, \qquad 0 \leq x \leq 9.
\]
\begin{parts}
  \item Beräkna $U(0)$, $U(1)$ och $U(9)$.
  \item Bestäm värdemängden $V_U$ för det angivna intervallet.
  \item Vid vilket $x$ blir $U = 8\,\text{V}$?
  \item Kan $U$ någonsin nå $12\,\text{V}$? Motivera.
\end{parts}

\uppgift{3.13}{Urladdning av en kondensator}
Spänningen över en urladdande kondensator ges av
\[
  u(t) = 10 \cdot e^{-t/2}\,\text{V}, \qquad t \geq 0,
\]
där $t$ anges i sekunder.
\begin{parts}
  \item Beräkna $u(0)$.
  \item Beräkna $u(2)$ och $u(4)$.
  \item Är funktionen växande eller avtagande?
  \item Bestäm värdemängden $V_u$ för $t \geq 0$.
  \item Blir $u(t)$ någonsin exakt noll? Motivera.
\end{parts}

\uppgift{3.14}{Sant eller falskt}
Avgör om påståendet är sant eller falskt och motivera kortfattat.
\begin{parts}
  \item Effekten $P(I) = RI^2$ blir negativ om strömmen är negativ.
  \item Strömmen $I(R) = \dfrac{12}{R}$ är definierad för alla $R$.
  \item Att $U(2) = 5$ betyder att $U$ har ett nollställe vid $I = 2\,\text{A}$.
  \item Om samma spänning $U$ har uppmätts med två olika strömmar $I$, är $I$ inte en funktion av $U$.
  \item Spänningen $u(t) = 3 \cdot 2^t$ (i V) har startvärdet $3\,\text{V}$.
\end{parts}
