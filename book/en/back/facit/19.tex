\facitavsnitt{ot}{Practice exam 1}{The whole course}
\begin{facit}

\svar{1} $I_1 = 3\,\text{mA}$, $I_2 = 2\,\text{mA}$

\svar{2} \sv{a} $R = 2\,\Omega$ or $R = 18\,\Omega$
\sv{b} $I = 3\,\text{A}$ and $1\,\text{A}$ respectively. $R = 18\,\Omega$ is preferable: the same
power with a third of the current, and the loss in the internal resistance is $6\,\text{W}$
instead of $54\,\text{W}$.

\svar{3} \sv{a} $U(I) = \frac{4(I + 3)}{I + 2}$, where $I \neq -2$ and $I \neq 3$
\sv{b} $U(2) = 5\,\text{V}$

\svar{4} \sv{a} $u(3) \approx 6.43\,\text{V}$
\sv{b} $0 \leq t \leq 10\,\text{s}$ and $2.30\,\text{V} \leq u(t) \leq 10\,\text{V}$
\sv{c} After about $6.23\,\text{s}$.

\svar{5} \sv{a} $u(t) = 5\sin\bigl(200\pi t + \frac{\pi}{2}\bigr)\,\text{V}$
\sv{b} The period is $10\,\text{ms}$. The curve starts at $5\,\text{V}$, is zero at $2.5$ and
$7.5\,\text{ms}$ and reaches $-5\,\text{V}$ at $5\,\text{ms}$.

\svar{6} \sv{a} $P'(I) = 24 - 12I$ \sv{b} $I = 2\,\text{A}$
\sv{c} Maximum, since $P''(2) = -12 < 0$.
\sv{d} The largest value is $P(2) = 24\,\text{W}$.
\sv{e} A downward-opening parabola with its maximum at $(2, 24)$. It crosses the $I$-axis at
$(0, 0)$ and $(4, 0)$.
\sv{f} $U = 12\,\text{V}$ and $R = 6\,\Omega$, so $R = R_{\text{i}}$ (maximum power transfer).

\svar{7} \sv{a} $I(t) = 0.01t^3 + 0.05t^2 + 0.5t + C$
\sv{b} $q(t) = 0.01t^3 + 0.05t^2 + 0.5t$
\sv{c} $q_{\text{tot}}(t) = 0.01t^3 + 0.05t^2 + 0.5t + 1$
\sv{d} Exactly $4.44\,\text{C}$

\svar{8} \sv{a} $I = -0.25 - j1.75\,\text{A} \approx 1.8\angle{-1.7}\,\text{A}$, with the angle
in radians ($-98.1^{\circ}$).
\sv{b} $R = 2\,\Omega$ and $X = 2\,\Omega$, so inductive.
\sv{c} $L = \frac{1}{50\pi}\,\text{H} \approx 6.37\,\text{mH}$

\svar{9} \sv{a} An arrow from the origin to $10 - j5$, in the fourth quadrant.
\sv{b} $I \approx 11.2e^{-j0.46}\,\text{mA}$: $\abs{I} = \sqrt{125} \approx 11.2\,\text{mA}$
and $\delta \approx -0.46\,\text{rad}$ ($-26.6^{\circ}$)
\sv{c} $w = 20\pi \approx 62.8\,\text{rad/s}$
\sv{d} $i(t) \approx 11.2e^{j(20\pi t - 0.46)}\,\text{mA}$

\svar{10} \sv{a} $U_1 \approx 1.81 + j0.85\,\text{V}$, $U_2 \approx 3.54 - j3.54\,\text{V}$,
$U_3 \approx 1.21 + j0.88\,\text{V}$
\sv{b} $U_{\text{tot}} \approx 6.56 - j1.81\,\text{V}$
\sv{c} $U_1$ and $U_3$ in the first quadrant, $U_2$ and $U_{\text{tot}}$ in the fourth;
$U_{\text{tot}}$ lies just below the real axis.
\sv{d} $u_{\text{tot}}(t) \approx 6.81\sin(wt - 15.4^{\circ})\,\text{V}$

\end{facit}

\facitavsnitt{ot2}{Practice exam 2}{The whole course}
\begin{facit}

\svar{1} $I_1 = 2\,\text{mA}$, $I_2 = 4\,\text{mA}$

\svar{2} \sv{a} $U = 4\,\text{V}$ or $U = 16\,\text{V}$
\sv{b} $U = 16\,\text{V}$: $I = 1\,\text{A}$, $R = 16\,\Omega$. $U = 4\,\text{V}$: $I = 4\,\text{A}$,
$R = 1\,\Omega$. $U = 16\,\text{V}$ is preferable: the same power with a quarter of the current.

\svar{3} \sv{a} $U(I) = \frac{3(I + 2)}{I + 3}$, where $I \neq \pm 3$ \sv{b} $I = 7\,\text{A}$

\svar{4} \sv{a} $RC = 10\,\text{s}$ \sv{b} $U_0 = 6.0 \cdot e^{0.5} \approx 9.89\,\text{V}$
\sv{c} $t = 10 \ln 2 \approx 6.93\,\text{s}$, independent of $U_0$.

\svar{5} \sv{a} Amplitude $325\,\text{V}$, RMS value $\approx 230\,\text{V}$, $f = 50\,\text{Hz}$,
$T = 20\,\text{ms}$ and phase $30^{\circ}$.
\sv{b} The curve starts at $162.5\,\text{V}$, has its maximum at $3.33\,\text{ms}$, zeros at
$8.33$ and $18.33\,\text{ms}$ and its minimum at $13.33\,\text{ms}$.

\svar{6} \sv{a} $i'(t) = 3t^2 - 18t + 24$ \sv{b} $t = 2\,\text{s}$ and $t = 4\,\text{s}$
\sv{c} Maximum at $t = 2$ ($i''(2) = -6$), minimum at $t = 4$ ($i''(4) = 6$).
\sv{d} $i(2) = 20\,\text{mA}$, $i(4) = 16\,\text{mA}$
\sv{e} The graph starts at the origin, rises to $(2, 20)$, falls to $(4, 16)$ and rises to
$(5, 20)$.
\sv{f} $u(1) = 0.5 \cdot 0.009 = 4.5\,\text{mV}$

\svar{7} \sv{a} $I(t) = 0.02t^3 + 0.2t^2 + t + C$
\sv{b} $q(t) = 0.02t^3 + 0.2t^2 + t$ \sv{c} $9.54\,\text{C}$ \sv{d} $3.18\,\text{A}$

\svar{8} \sv{a} $Z = 30 + j40 \approx 50\angle 53.1^{\circ}\,\Omega$
\sv{b} $I \approx 2\angle{-53.1^{\circ}}\,\text{A} = 1.2 - j1.6\,\text{A}$
\sv{c} $L = \frac{2}{5\pi}\,\text{H} \approx 127\,\text{mH}$

\svar{9} \sv{a} An arrow from the origin to $(-8, -6)$, in the third quadrant.
\sv{b} $I \approx 10e^{-j2.50}\,\text{mA}$ ($\delta \approx -143.1^{\circ}$)
\sv{c} $\omega = 100\pi\,\text{rad/s}$ \sv{d} $i(t) \approx 10\sin(100\pi t - 2.50)\,\text{mA}$

\svar{10} \sv{a} $U_1 \approx 2.00 + j3.46\,\text{V}$, $U_2 \approx 2.60 - j1.50\,\text{V}$,
$U_3 \approx -1.73 + j1.00\,\text{V}$
\sv{b} $U_{\text{tot}} \approx 2.87 + j2.96\,\text{V}$
\sv{c} $U_1$ in the first, $U_2$ in the fourth and $U_3$ in the second quadrant; $U_{\text{tot}}$ in
the first.
\sv{d} $u_{\text{tot}}(t) \approx 4.12\sin(wt + 46.0^{\circ})\,\text{V}$

\end{facit}
