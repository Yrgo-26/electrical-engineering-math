\facitavsnitt{c3}{Chapter 3}{Linear equations and inequalities}
\begin{facit}

\svar{1.1} \sv{a} $4R + 14$ \sv{b} $U^2 + 6U = U(U + 6)$ \sv{c} $3I + 2$

\svar{1.2} \sv{a} $(R + 5)(R - 5)$ \sv{b} $3U(U + 2)$ \sv{c} $(I - 4)^2$

\svar{2.1} \sv{a} $15 = R \cdot 0.3$ gives $R = 50\,\Omega$.
\sv{b} $50 \times 0.3 = 15\,\text{V}$, which is correct.

\svar{2.2} \sv{a} $24 = (R_1 + 40) \times 0.4$ gives $R_1 + 40 = 60$, so $R_1 = 20\,\Omega$.
\sv{b} The same answer, $R_1 = 20\,\Omega$.
\sv{c} $I = \frac{24}{20 + 40} = 0.4\,\text{A}$, which is correct.

\svar{2.3} \sv{a} $3 = \frac{3}{R_1 + 3} \times 9$ gives $R_1 = 6\,\Omega$.
\sv{b} $\frac{3}{6 + 3} \times 9 = 3\,\text{V}$, which is correct.

\svar{2.4} $R = \frac{\tau}{C} = 500\,\Omega$

\svar{2.5} \sv{a} $U_{\text{CE}} \cdot I_{\text{C}} \leq P_{\max}$, that is,
$U_{\text{CE}} \cdot 0.05 \leq 0.5$ \sv{b} $U_{\text{CE}} \leq 10\,\text{V}$

\svar{2.6} \sv{a} $10.4\,\text{V}$
\sv{b} $12 - 0.8I \geq 9$ gives $I \leq 3.75\,\text{A}$; the sign reverses on dividing by
$-0.8$.
\sv{c} $12 - 0.8I = 13 - 1.0I$ gives $I = 5\,\text{A}$, where both give $8\,\text{V}$.
\sv{d} The first battery: $5.6\,\text{V}$ against $5.0\,\text{V}$.

\svar{2.7} \sv{a} $-0.5\,\text{V} \leq \delta \leq 0.5\,\text{V}$
\sv{b} $4.5\,\text{V} \leq u \leq 5.5\,\text{V}$

\svar{2.8} \sv{a} $5\,\text{A}$ \sv{b} $I = 2\,\text{A}$ and $I = 8\,\text{A}$
\sv{c} $2\,\text{A} < I < 8\,\text{A}$

\svar{3.1} \sv{a} $x = 5$ \sv{b} $x = 9$ \sv{c} $x = 5$ \sv{d} $x = 24$ \sv{e} $x = 4$
\sv{f} $x = -6$

\svar{3.2} \sv{a} $x = 4$ \sv{b} $x = 2$ \sv{c} $x = 5$
\sv{d} Every $x$: the two sides are identical, so the equation is an identity.
\sv{e} No solution: $x$ cancels and what is left is $8 = 5$, which is false.

\svar{3.3} \sv{a} $x = 6$ \sv{b} $x = 8$ \sv{c} $x = 10$ \sv{d} $x = 3$

\svar{3.4} \sv{a} $x = 6$ \sv{b} $x = 15$ \sv{c} $R_1 = 1\,\text{k}\Omega$
\sv{d} $\frac{6}{0.2} = \frac{U}{0.5}$ gives $U = 15\,\text{V}$.

\svar{3.5} \sv{a} $I = \frac{U}{R}$ \sv{b} $R = \frac{P}{I^2}$ \sv{c} $R_2 = R_{\text{TOT}} - R_1$
\sv{d} $C = \frac{\tau}{R}$ \sv{e} $R_{\text{i}} = \frac{E - U}{I}$
\sv{f} $P_{\text{in}} = \frac{P_{\text{out}}}{\eta}$

\svar{3.6} \sv{a} $x < 5$: open circle at $5$, arrow to the left.
\sv{b} $x \geq 4$: filled circle at $4$, arrow to the right.
\sv{c} $x < -4$: open circle at $-4$, arrow to the left.
\sv{d} $x \geq 3$: filled circle at $3$, arrow to the right.
\sv{e} $x < -3$: open circle at $-3$, arrow to the left.

\svar{3.7} \sv{a} $1 < x < 6$ \sv{b} $-1 \leq x \leq 3$
\sv{c} $5I \leq 0.25$ gives $I \leq 0.05\,\text{A} = 50\,\text{mA}$.
\sv{d} $4.0 \leq x \leq 6.6$

\svar{3.8} \sv{a} $x = 6$ or $x = -6$ \sv{b} $x = 8$ or $x = -2$
\sv{c} $x = 3$ or $x = -4$ \sv{d} $x = -4$, a single solution.
\sv{e} No solution: an absolute value is never negative.

\svar{3.9} \sv{a} $-2 \leq x - 5 \leq 2$, so $3 \leq x \leq 7$. \sv{b} $1 < x < 5$
\sv{c} $\abs{u - 5} \leq 0.1$, so $4.9\,\text{V} \leq u \leq 5.1\,\text{V}$.
\sv{d} No: $\abs{1015 - 1000} = 15 > 10$.

\svar{3.10} \sv{a} $120\,\Omega$ \sv{b} $45\,{}^{\circ}\text{C}$
\sv{c} At most $95\,{}^{\circ}\text{C}$, since $\Delta T \leq 75\,{}^{\circ}\text{C}$.
\sv{d} $\Delta T = \frac{R - R_0}{R_0\,\alpha}$

\svar[Test yourself]{T} \sv{1} $x = 8$; both sides are then $24$.
\sv{2} $I = 0.03\,\text{A} = 30\,\text{mA}$ \sv{3} $I \leq 6\,\text{A}$
