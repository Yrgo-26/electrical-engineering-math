\section{Summary}
\label{c14:sec:sammanfattning}

\begin{booktable}{@{}p{12.5em}L@{}}
  \thead{Concept} & \thead{Formula} \\
  \midrule
  Antiderivative & $F'(x) = f(x)$ \\
  Indefinite integral & $\int f(x)\,dx = F(x) + C$ \\
  Definite integral & $\int_a^b f(x)\,dx = F(b) - F(a)$ \\
  $\int x^n\,dx$ & $\frac{x^{n+1}}{n+1} + C$ \\
  $\int e^{kx}\,dx$ & $\frac{1}{k}e^{kx} + C$ \\
  Charge & $q(t) = \int i(t)\,dt$ \\
\end{booktable}

\needspace{6\baselineskip}
\subsection*{What you should be able to do}
After this chapter, you should be able to:
\begin{itemize}
  \item Understand what an integral is.
  \item Understand the concept of an antiderivative and find one for simple functions.
  \item Carry out simple integrations.
\end{itemize}

\testadigsjalv
\begin{enumerate}
  \item Calculate $\displaystyle\int_0^2 (3x^2 + 1)\,dx$.
  \item Find the antiderivative $F(x)$ of $f(x) = 4x + e^x$. Include the constant of
    integration $C$ in your answer.
\end{enumerate}
