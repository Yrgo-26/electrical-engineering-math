\section{Summary}
\label{c9:sec:sammanfattning}

\begin{booktable}{@{}p{12.5em}L@{}}
  \thead{Concept} & \thead{Formula} \\
  \midrule
  Degrees to radians & $v_{\text{rad}} = v_{\text{deg}} \cdot \pi/180^{\circ}$ \\
  Angular frequency & $\omega = 2\pi f = 2\pi/T$ \\
  Period & $T = 1/f$ \\
  AC voltage equation & $u(t) = \abs{U}\sin(\omega t + \delta)$ \\
\end{booktable}

\needspace{6\baselineskip}
\subsection*{What you should be able to do}
After this chapter, you should be able to:
\begin{itemize}
  \item Recognise the trigonometric functions and their parameters.
  \item Use the angle measures degrees and radians and convert between them.
\end{itemize}

\testadigsjalv
\begin{enumerate}
  \item Convert $150^{\circ}$ to radians and $\dfrac{5\pi}{6}$ to degrees.
  \item An AC voltage has the amplitude $5\,\text{V}$, the frequency $50\,\text{Hz}$ and the phase
    $\pi/4\,\text{rad}$. Write the voltage's equation $u(t)$.
\end{enumerate}
