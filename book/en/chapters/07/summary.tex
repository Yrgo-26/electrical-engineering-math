\section{Summary}
\label{c7:sec:sammanfattning}

\begin{booktable}{@{}p{12.5em}L@{}}
  \thead{Concept} & \thead{Formula} \\
  \midrule
  Magnitude & $\abs{\mathbf{u}} = \sqrt{u_x^2 + u_y^2}$ \\
  Angle with the $x$-axis & $v = \arctan(u_y / u_x)$ + quadrant correction \\
  Scalar product & $\mathbf{u} \cdot \mathbf{v} = u_x v_x + u_y v_y$ \\
  Angle between vectors &
    $\cos \alpha = \dfrac{\mathbf{u} \cdot \mathbf{v}}{\abs{\mathbf{u}}\abs{\mathbf{v}}}$ \\
  Addition & $\mathbf{u} + \mathbf{v} = (u_x + v_x, u_y + v_y)$ \\
  Scalar multiplication & $k\mathbf{u} = (ku_x, ku_y)$ \\
\end{booktable}

\subsection*{What you should be able to do}
After this chapter, you should be able to:
\begin{itemize}
  \item Understand the concept of a vector and how a vector is represented mathematically.
  \item Carry out simple calculations with vectors.
\end{itemize}

\testadigsjalv
\begin{enumerate}
  \item A vector is $\mathbf{u} = (3, -4)$. Calculate the magnitude $\abs{\mathbf{u}}$ and the
    angle with the $x$-axis.
  \item Calculate $\mathbf{w} = 2\mathbf{u} - \mathbf{v}$ for $\mathbf{u} = (2, 1)$ and
    $\mathbf{v} = (-1, 3)$.
\end{enumerate}
