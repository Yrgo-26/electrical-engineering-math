\uppgiftsdel{Part 1}{Review exercises}

\uppgift{1.1}{Sensor module on a railway wagon}
A sensor module on a railway wagon is supplied from the wagon's $12\,\text{V}$ system. To limit
the current in the event of a short circuit, a protective resistor of $4\,\text{k}\Omega$ sits in
series with the module. A measurement shows that the sensor module consumes $5\,\text{mW}$.

\bild[0.5\linewidth]{lectures/L07/appendix/images/1.1_circuit_en.png}

You want to find out how large the current $I$ through the circuit is, but have no ammeter to
hand, only your knowledge of circuit theory.
\begin{parts}
  \item The voltage across the protective resistor is $4I$ (with $I$ in mA,
    $\text{k}\Omega \cdot \text{mA} = \text{V}$). Show that the voltage across the sensor module
    is $12 - 4I$ and that the power relation leads to the equation
    \[
      4I^2 - 12I + 5 = 0.
    \]
  \item Solve the equation for the current $I$.
  \item The equation gives two solutions. Calculate the voltage across the sensor module and the
    power loss in the protective resistor in both cases. Which operating point is the most
    reasonable for a sensor module, and why?
\end{parts}

\uppgiftsdel{Part 2}{New material}
In AC circuits, voltages and currents are represented by vectors, known as \textbf{phasors}: the
length of the vector gives the magnitude, and its angle with the positive $x$-axis gives the phase
angle. Two phasors with the same direction are \emph{in phase}, and two phasors with opposite
directions are \emph{in antiphase}.

\uppgift{2.1}{Adding voltages}
Two AC voltages are represented by the phasors $\mathbf{U}_1 = (3, 2)\,\text{V}$ and
$\mathbf{U}_2 = (4, 6)\,\text{V}$. Find the coordinates of:
\begin{delar}(4)
  \task $\mathbf{U}_1 + \mathbf{U}_2$
  \task $2\mathbf{U}_1 + 3\mathbf{U}_2$
  \task $\mathbf{U}_1 - \mathbf{U}_2$
  \task $-2\mathbf{U}_1 - 4\mathbf{U}_2$
\end{delar}

\uppgift{2.2}{Calculating with three currents}
Three alternating currents are represented by the phasors $\mathbf{I}_a = (1, 4)\,\text{A}$,
$\mathbf{I}_b = (-4, 2)\,\text{A}$ and $\mathbf{I}_c = (3, -4)\,\text{A}$. Find the
coordinates of:
\begin{delar}(4)
  \task $\mathbf{I}_a + \mathbf{I}_b + \mathbf{I}_c$
  \task $\mathbf{I}_a - \mathbf{I}_b - \mathbf{I}_c$
  \task $2\mathbf{I}_a + 4\mathbf{I}_b$
  \task $2\mathbf{I}_a - 4\mathbf{I}_b$
\end{delar}

\uppgift{2.3}{Magnitudes of voltages}
The voltages $\mathbf{U}_a = (2, 1)\,\text{V}$ and $\mathbf{U}_b = (1, -2)\,\text{V}$ are
given. Find, in exact form, the magnitude of:
\begin{delar}(4)
  \task $\mathbf{U}_a$
  \task $3\mathbf{U}_a + \mathbf{U}_b$
  \task $\mathbf{U}_a - \mathbf{U}_b$
  \task $3\mathbf{U}_a - 4\mathbf{U}_b$
\end{delar}

\uppgift{2.4}{Solve the equations}
Find $x$ and $y$:
\begin{delar}(2)
  \task $(x, y) = 5(3, 1) + 2(4, -5)$
  \task $(15, 12) = (x, 10) + (23, y)$
\end{delar}

\uppgift{2.5}{Currents in phase and in antiphase}
If $\mathbf{I}_1 = k \cdot \mathbf{I}_2$, the phasors $\mathbf{I}_1$ and $\mathbf{I}_2$ are
parallel: in phase if $k > 0$ and in antiphase if $k < 0$. Which of the following currents are
parallel to $\mathbf{I} = (2, -3)\,\text{A}$?
\begin{delar}(4)
  \task $(6, -9)\,\text{A}$
  \task $(-8, 16)\,\text{A}$
  \task $(10, 15)\,\text{A}$
  \task $(-12, 18)\,\text{A}$
\end{delar}

\uppgift{2.6}{Finding unknown coefficients}
The voltages $\mathbf{U}_1 = (-2, 3)\,\text{V}$ and $\mathbf{U}_2 = (3, -1)\,\text{V}$ are
given. Find the numbers $a$ and $b$ such that
\[
  a\mathbf{U}_1 + \mathbf{U}_2 = (0, b)\,\text{V}.
\]

\uppgift{2.7}{Phase shift between voltage and current}
The voltage across a load is $\mathbf{U} = (4, 4)\,\text{V}$ and the current through it is
$\mathbf{I} = (-1, 2)\,\text{A}$.
\begin{parts}
  \item Calculate the magnitudes $\abs{\mathbf{U}}$ and $\abs{\mathbf{I}}$.
  \item Calculate the phase angles $v_U$ and $v_I$ with the positive $x$-axis.
  \item Calculate the phase shift, that is, the angle $\alpha$ between $\mathbf{U}$ and $\mathbf{I}$.
\end{parts}

\uppgiftsdel{Part 3}{Extra exercises}
The exercises below are extra practice, best done on your own after the lesson.

\uppgift{3.1}{Distance between points}
Calculate the distance between the points:
\begin{delar}(3)
  \task $(0, 0)$ and $(3, 4)$
  \task $(1, 2)$ and $(4, 6)$
  \task $(-2, 1)$ and $(2, -2)$
  \task $(5, -1)$ and $(5, 7)$
  \task $(-3, 2)$ and $(1, 2)$
\end{delar}

\uppgift{3.2}{The vector between two points}
The vector from the point $A$ to the point $B$ is given by $\overrightarrow{AB} = (x_B - x_A;\,
y_B - y_A)$.
\begin{parts}
  \item $A = (1, 2)$ and $B = (5, 5)$. Find $\overrightarrow{AB}$ and $\abs{\overrightarrow{AB}}$.
  \item $A = (-1, 4)$ and $B = (2, 0)$. Find $\overrightarrow{AB}$ and $\abs{\overrightarrow{AB}}$.
  \item Find $\overrightarrow{BA}$ in a) and compare it with $\overrightarrow{AB}$.
  \item What holds in general between $\overrightarrow{AB}$ and $\overrightarrow{BA}$?
\end{parts}

\uppgift{3.3}{The magnitude of a voltage}
Find the magnitude $\abs{\mathbf{U}}$ of the voltage in exact form and as an approximation to two
decimal places.
\begin{delar}(3)
  \task $\mathbf{U} = (6, 8)\,\text{V}$
  \task $\mathbf{U} = (-5, 12)\,\text{V}$
  \task $\mathbf{U} = (1, 1)\,\text{V}$
  \task $\mathbf{U} = (-3, -3)\,\text{V}$
  \task $\mathbf{U} = (0, -7)\,\text{V}$
\end{delar}

\uppgift{3.4}{Phase angle}
Find the phase angle $v$ of the current with the positive $x$-axis. State which quadrant the
phasor lies in and whether a quadrant correction is needed.
\begin{delar}(3)
  \task $\mathbf{I} = (1, 1)\,\text{A}$
  \task $\mathbf{I} = (-1, 1)\,\text{A}$
  \task $\mathbf{I} = (-1, -1)\,\text{A}$
  \task $\mathbf{I} = (1, -1)\,\text{A}$
  \task $\mathbf{I} = (4, 3)\,\text{A}$
\end{delar}

\uppgift{3.5}{From magnitude and phase angle to components}
A voltage with the magnitude $\abs{\mathbf{U}}$ and the phase angle $v$ with the positive $x$-axis
has the components
\[
  U_x = \abs{\mathbf{U}} \cos v, \qquad U_y = \abs{\mathbf{U}} \sin v.
\]
Find the components:
\begin{delar}(2)
  \task $\abs{\mathbf{U}} = 10\,\text{V}$, $v = 30^{\circ}$
  \task $\abs{\mathbf{U}} = 5\,\text{V}$, $v = 90^{\circ}$
  \task $\abs{\mathbf{U}} = 4\,\text{V}$, $v = 180^{\circ}$
  \task $\abs{\mathbf{U}} = 2\,\text{V}$, $v = 225^{\circ}$
  \task* Check in a) that $\sqrt{U_x^2 + U_y^2} = 10\,\text{V}$.
\end{delar}

\uppgift{3.6}{Phase shift}
First calculate the phase angles $v_U$ and $v_I$ with the positive $x$-axis, and then the phase
shift $\alpha$, that is, the angle between the voltage and the current.
\begin{delar}(2)
  \task $\mathbf{U} = (1, 0)\,\text{V}$, $\mathbf{I} = (1, 1)\,\text{A}$
  \task $\mathbf{U} = (3, 0)\,\text{V}$, $\mathbf{I} = (0, 4)\,\text{A}$
  \task $\mathbf{U} = (2, 1)\,\text{V}$, $\mathbf{I} = (-1, 3)\,\text{A}$
  \task $\mathbf{U} = (2, 2)\,\text{V}$, $\mathbf{I} = (-3, -3)\,\text{A}$
\end{delar}

\uppgift{3.7}{Resistor and inductor in series}
Across a resistor and an inductor in series lie the voltages $\mathbf{U}_R = (12, 0)\,\text{V}$
and $\mathbf{U}_L = (0, 5)\,\text{V}$.
\begin{parts}
  \item Calculate the total voltage $\mathbf{U} = \mathbf{U}_R + \mathbf{U}_L$.
  \item Calculate the magnitude $\abs{\mathbf{U}}$.
  \item Calculate the phase angle with the positive $x$-axis.
  \item Why is $\abs{\mathbf{U}}$ not equal to $12 + 5 = 17\,\text{V}$?
\end{parts}

\uppgift{3.8}{Resistor, inductor and capacitor in series}
In a series circuit, the voltage $\mathbf{U}_R = (30, 0)\,\text{V}$ lies across the resistor,
$\mathbf{U}_L = (0, 70)\,\text{V}$ across the inductor and $\mathbf{U}_C = (0, -30)\,\text{V}$
across the capacitor.
\begin{parts}
  \item Calculate the total voltage $\mathbf{U} = \mathbf{U}_R + \mathbf{U}_L + \mathbf{U}_C$.
  \item Calculate the magnitude $\abs{\mathbf{U}}$.
  \item Calculate the phase angle with the positive $x$-axis.
  \item What is the total voltage if the capacitor voltage is instead $(0, -70)\,\text{V}$? What does
    the result mean?
\end{parts}

\uppgift{3.9}{Unit vectors}
A unit vector has length $1$ and is obtained as $\hat{\mathbf{u}} =
\dfrac{\mathbf{u}}{\abs{\mathbf{u}}}$. The current $\mathbf{I} = (3, 4)\,\text{A}$ is given.
\begin{parts}
  \item Find the unit vector in the direction of the current.
  \item Check that the answer in a) really has length $1$.
  \item Find the unit vector in the direction $(-6, 8)$.
  \item Find a current with the magnitude $15\,\text{A}$ that is in phase with $\mathbf{I}$.
  \item Find a current with the magnitude $15\,\text{A}$ that is in antiphase with $\mathbf{I}$.
\end{parts}

\uppgift{3.10}{Three-phase system}
In a three-phase system the three phase voltages have the same magnitude but different phase
angles: $\mathbf{U}_1 = (230, 0)\,\text{V}$, $\mathbf{U}_2 = (-115, -199)\,\text{V}$ and
$\mathbf{U}_3 = (-115, 199)\,\text{V}$.
\begin{parts}
  \item Calculate the sum $\mathbf{U}_1 + \mathbf{U}_2 + \mathbf{U}_3$.
  \item What does the result mean?
  \item Calculate the magnitude and the phase angle of $\mathbf{U}_2$.
  \item Calculate the line voltage $\mathbf{U}_{12} = \mathbf{U}_1 - \mathbf{U}_2$ and its magnitude.
\end{parts}

\uppgift{3.11}{In phase and in antiphase}
The voltage $\mathbf{U} = (4, -6)\,\text{V}$ is given.
\begin{parts}
  \item Which of $(2, -3)\,\text{V}$, $(-8, 12)\,\text{V}$, $(6, -4)\,\text{V}$ and $(-2;\,
    3)\,\text{V}$ are parallel to $\mathbf{U}$?
  \item Which of them are in antiphase with $\mathbf{U}$?
  \item Find $t$ such that $(t, 9)\,\text{V}$ is parallel to $\mathbf{U}$.
  \item Can a phasor be parallel to itself? Justify your answer.
\end{parts}

\uppgift{3.12}{Resistor and capacitor in parallel}
Through a resistor and a capacitor connected in parallel flow the currents $\mathbf{I}_R = (4;\,
0)\,\text{A}$ and $\mathbf{I}_C = (0, 3)\,\text{A}$.
\begin{parts}
  \item Calculate the total current $\mathbf{I} = \mathbf{I}_R + \mathbf{I}_C$.
  \item Calculate the magnitude $\abs{\mathbf{I}}$.
  \item Calculate the phase angle with the positive $x$-axis.
  \item How large must the capacitor current be for the phase angle to become $45^{\circ}$?
\end{parts}

\uppgift{3.13}{The power triangle}
The power in an AC circuit can be described by the vector $\mathbf{S} = (P, Q)$, where $P$ is
the active power in watts (W) and $Q$ is the reactive power in var. The magnitude
$\abs{\mathbf{S}}$ of the vector is the apparent power in volt-amperes (VA). A motor has $P =
800\,\text{W}$ and $Q = 600\,\text{var}$.
\begin{parts}
  \item Calculate the apparent power $\abs{\mathbf{S}}$.
  \item Calculate the angle $v$ of the vector with the positive $x$-axis.
  \item The power factor is $\cos v$. Calculate it.
  \item A capacitor is connected and contributes the vector $(0, -600)$. Find the new vector
    $\mathbf{S}$ and the new apparent power.
\end{parts}

\uppgift{3.14}{Find the error}
Each line contains a common vector error. Explain the error and give the correct answer.
\begin{parts}
  \item $\abs{(3, 4)} = 3 + 4 = 7$
  \item $2(3, -1) = (6, -1)$
  \item Two phasors have the angles $30^{\circ}$ and $100^{\circ}$ with the positive $x$-axis, so the
    angle between them is $130^{\circ}$.
  \item The vector $(-3, 4)$ makes the angle $\arctan\!\left(\dfrac{4}{-3}\right) \approx
    -53.1^{\circ}$ with the positive $x$-axis.
  \item $\abs{-2\mathbf{u}} = -2\abs{\mathbf{u}}$
\end{parts}
