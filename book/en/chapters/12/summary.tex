\section{Summary}
\label{c12:sec:sammanfattning}

\begin{booktable}{@{}p{12.5em}L@{}}
  \thead{Concept} & \thead{Formula or rule} \\
  \midrule
  Derivative of $x^n$ & $nx^{n-1}$ \\
  Stationary point & $f'(x_0) = 0$ \\
  Maximum & $f'(x_0) = 0$ and $f''(x_0) < 0$ \\
  Minimum & $f'(x_0) = 0$ and $f''(x_0) > 0$ \\
  Equation of the tangent & $y = f'(x_0)(x - x_0) + f(x_0)$ \\
\end{booktable}

\needspace{6\baselineskip}
\subsection*{What you should be able to do}
After this chapter, you should be able to:
\begin{itemize}
  \item Understand how the derivative relates to the tangent, the equation of a straight line,
    and maximum and minimum points.
  \item Differentiate polynomial functions.
\end{itemize}

\testadigsjalv
\begin{enumerate}
  \item Differentiate $f(x) = 4x^3 - 2x^2 + 7$.
  \item Find the coordinates of the turning point of $f(x) = x^2 - 2x - 8$ and decide whether it
    is a maximum or a minimum.
\end{enumerate}
